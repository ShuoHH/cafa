cmake_minimum_required(VERSION 3.22)
project(FrozenPlanFileProject LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(ALP_ROOT "" CACHE PATH "External ALP source root")
set(ALP_LIBRARY "" CACHE FILEPATH "Already built ALP library")
set(STREAMING_HEADERS_ROOT "" CACHE PATH "External NeaTS benchmark header root")
if(NOT EXISTS "${ALP_ROOT}/include/alp.hpp")
  message(FATAL_ERROR "Set ALP_ROOT to external ALP source with include/alp.hpp")
endif()
if(NOT ALP_LIBRARY OR NOT EXISTS "${ALP_LIBRARY}")
  message(FATAL_ERROR "Set ALP_LIBRARY to the existing built libALP.so (no ALP download/build is performed here)")
endif()
foreach(header Gorilla/CompressorGorilla.hpp Chimp/CompressorChimp.hpp
               Chimp128/CompressorChimp128.hpp TSXor/CompressorTSXor.hpp
               lib/BitStream.hpp)
  if(NOT EXISTS "${STREAMING_HEADERS_ROOT}/${header}")
    message(FATAL_ERROR "Set STREAMING_HEADERS_ROOT to external NeaTS benchmark root containing ${header}")
  endif()
endforeach()
add_executable(FrozenPlanFile frozen_plan.cpp)
target_include_directories(FrozenPlanFile PRIVATE "${ALP_ROOT}/include" "${REPO}/harness" "${STREAMING_HEADERS_ROOT}")
target_link_libraries(FrozenPlanFile PRIVATE "${ALP_LIBRARY}")
if(UNIX)
  get_filename_component(ALP_LIBRARY_DIR "${ALP_LIBRARY}" DIRECTORY)
  set_target_properties(FrozenPlanFile PROPERTIES BUILD_RPATH "${ALP_LIBRARY_DIR}" INSTALL_RPATH "${ALP_LIBRARY_DIR}")
  target_link_libraries(FrozenPlanFile PRIVATE m ${CMAKE_DL_LIBS})
endif()
